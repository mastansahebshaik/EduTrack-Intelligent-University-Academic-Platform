package citationflow;

public class FlowEdge {

    int from;
    int to;

    int capacity;
    int flow;

    FlowEdge reverse;

    public FlowEdge(int from, int to, int capacity) {
        this.from = from;
        this.to = to;
        this.capacity = capacity;
        this.flow = 0;
    }

    public int residualCapacity() {
        return capacity - flow;
    }

    public void addFlow(int amount) {
        flow += amount;
        reverse.flow -= amount;
    }
}

package citationflow;

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class CitationGraph {

    private int[] paperIds;
    private int paperCount;

    private int[] sourceIds;
    private int[] targetIds;

    private int edgeCount;

    public CitationGraph() {
        paperIds = new int[30000];

        sourceIds = new int[10000];
        targetIds = new int[10000];

        paperCount = 0;
        edgeCount = 0;
    }

    public void load(String fileName) throws IOException {

        BufferedReader br =
                new BufferedReader(new FileReader(fileName));

        String line;

        while ((line = br.readLine()) != null) {

            line = line.trim();

            if (line.length() == 0) {
                continue;
            }

            String[] parts = line.split("\\s+");

            if (parts.length < 2) {
                continue;
            }

            int source = Integer.parseInt(parts[0]);
            int target = Integer.parseInt(parts[1]);

            addPaper(source);
            addPaper(target);

            sourceIds[edgeCount] = source;
            targetIds[edgeCount] = target;

            edgeCount++;
        }

        br.close();

        System.out.println();
        System.out.println("Citation dataset loaded successfully.");
        System.out.println("Number of papers    : " + paperCount);
        System.out.println("Number of citations : " + edgeCount);
    }

    private void addPaper(int id) {

        if (getInternalIndex(id) != -1) {
            return;
        }

        paperIds[paperCount] = id;
        paperCount++;
    }

    public int getInternalIndex(int paperId) {

        for (int i = 0; i < paperCount; i++) {

            if (paperIds[i] == paperId) {
                return i;
            }
        }

        return -1;
    }

    public int getPaperId(int index) {

        if (index < 0 || index >= paperCount) {
            return -1;
        }

        return paperIds[index];
    }

    public int getPaperCount() {
        return paperCount;
    }

    public int getEdgeCount() {
        return edgeCount;
    }

    public int getSourceId(int edge) {
        return sourceIds[edge];
    }

    public int getTargetId(int edge) {
        return targetIds[edge];
    }

    public void displayInformation() {

        System.out.println();
        System.out.println("==============================================");
        System.out.println("          CITATION GRAPH INFORMATION");
        System.out.println("==============================================");

        System.out.println("Dataset              : Cora");
        System.out.println("Number of papers     : " + paperCount);
        System.out.println("Number of citations  : " + edgeCount);
        System.out.println("Graph type           : Directed");
        System.out.println("Edge capacity        : 1");

        System.out.println("==============================================");
    }

    public void displayEdges(int limit) {

        System.out.println();
        System.out.println("==============================================");
        System.out.println("              CITATION EDGES");
        System.out.println("==============================================");

        int count = edgeCount;

        if (count > limit) {
            count = limit;
        }

        for (int i = 0; i < count; i++) {

            System.out.println(
                    sourceIds[i] + " -> " + targetIds[i]);
        }

        if (edgeCount > limit) {

            System.out.println(
                    "... " + (edgeCount - limit)
                    + " more citations");
        }

        System.out.println("==============================================");
    }

    public void displayPaperIds() {

        System.out.println();
        System.out.println("Available Paper IDs:");

        for (int i = 0; i < paperCount; i++) {

            System.out.print(paperIds[i] + " ");

            if ((i + 1) % 15 == 0) {
                System.out.println();
            }
        }

        System.out.println();
    }

    public int[] getPaperIds() {
        return paperIds;
    }

    public int[] getSourceIds() {
        return sourceIds;
    }

    public int[] getTargetIds() {
        return targetIds;
    }
}

package citationflow;

public class FordFulkerson {

    private CitationGraph graph;

    private int[][] capacity;
    private int[][] residual;

    private int vertices;

    private int augmentationCount;

    public FordFulkerson(CitationGraph graph) {

        this.graph = graph;

        vertices = graph.getPaperCount();

        capacity =
                new int[vertices][vertices];

        residual =
                new int[vertices][vertices];

        buildNetwork();
    }

    private void buildNetwork() {

        for (int i = 0;
             i < graph.getEdgeCount();
             i++) {

            int source =
                    graph.getInternalIndex(
                            graph.getSourceId(i));

            int target =
                    graph.getInternalIndex(
                            graph.getTargetId(i));

            capacity[source][target] += 1;
            residual[source][target] += 1;
        }
    }

    private boolean dfs(
            int current,
            int sink,
            boolean[] visited,
            int[] parent) {

        visited[current] = true;

        if (current == sink) {
            return true;
        }

        for (int next = 0;
             next < vertices;
             next++) {

            if (!visited[next]
                    && residual[current][next] > 0) {

                parent[next] = current;

                if (dfs(
                        next,
                        sink,
                        visited,
                        parent)) {

                    return true;
                }
            }
        }

        return false;
    }

    public FlowResult calculate(
            int source,
            int sink,
            boolean showPaths) {

        int maxFlow = 0;

        augmentationCount = 0;

        int[] parent =
                new int[vertices];

        long start =
                System.nanoTime();

        while (true) {

            boolean[] visited =
                    new boolean[vertices];

            for (int i = 0;
                 i < vertices;
                 i++) {

                parent[i] = -1;
            }

            boolean found =
                    dfs(
                            source,
                            sink,
                            visited,
                            parent);

            if (!found) {
                break;
            }

            int pathFlow =
                    Integer.MAX_VALUE;

            int current = sink;

            while (current != source) {

                int previous =
                        parent[current];

                if (residual[previous][current]
                        < pathFlow) {

                    pathFlow =
                            residual[previous][current];
                }

                current = previous;
            }

            augmentationCount++;

            if (showPaths) {

                System.out.print(
                        "\nAugmenting Path "
                        + augmentationCount
                        + ": ");

                printPath(
                        source,
                        sink,
                        parent);

                System.out.println(
                        "Bottleneck Capacity : "
                        + pathFlow);
            }

            current = sink;

            while (current != source) {

                int previous =
                        parent[current];

                residual[previous][current]
                        -= pathFlow;

                residual[current][previous]
                        += pathFlow;

                current = previous;
            }

            maxFlow += pathFlow;

            if (showPaths) {

                System.out.println(
                        "Current Maximum Flow : "
                        + maxFlow);
            }
        }

        long end =
                System.nanoTime();

        return new FlowResult(
                "Ford-Fulkerson",
                maxFlow,
                augmentationCount,
                end - start);
    }

    private void printPath(
            int source,
            int sink,
            int[] parent) {

        int[] path =
                new int[vertices];

        int count = 0;

        int current = sink;

        while (current != -1) {

            path[count++] = current;

            if (current == source) {
                break;
            }

            current = parent[current];
        }

        for (int i = count - 1;
             i >= 0;
             i--) {

            System.out.print(
                    graph.getPaperId(path[i]));

            if (i > 0) {
                System.out.print(" -> ");
            }
        }

        System.out.println();
    }

    public int[][] getResidual() {
        return residual;
    }

    public int[][] getCapacity() {
        return capacity;
    }

    public int getAugmentationCount() {
        return augmentationCount;
    }
}

package citationflow;

public class EdmondsKarp {

    private CitationGraph graph;

    private int[][] capacity;
    private int[][] residual;

    private int vertices;

    private int augmentationCount;

    public EdmondsKarp(CitationGraph graph) {

        this.graph = graph;

        vertices = graph.getPaperCount();

        capacity =
                new int[vertices][vertices];

        residual =
                new int[vertices][vertices];

        buildNetwork();
    }

    private void buildNetwork() {

        for (int i = 0;
             i < graph.getEdgeCount();
             i++) {

            int source =
                    graph.getInternalIndex(
                            graph.getSourceId(i));

            int target =
                    graph.getInternalIndex(
                            graph.getTargetId(i));

            capacity[source][target] += 1;
            residual[source][target] += 1;
        }
    }

    private boolean bfs(
            int source,
            int sink,
            int[] parent) {

        boolean[] visited =
                new boolean[vertices];

        int[] queue =
                new int[vertices];

        int front = 0;
        int rear = 0;

        queue[rear++] = source;

        visited[source] = true;

        while (front < rear) {

            int current =
                    queue[front++];

            for (int next = 0;
                 next < vertices;
                 next++) {

                if (!visited[next]
                        && residual[current][next] > 0) {

                    parent[next] = current;

                    visited[next] = true;

                    queue[rear++] = next;

                    if (next == sink) {
                        return true;
                    }
                }
            }
        }

        return false;
    }

    public FlowResult calculate(
            int source,
            int sink,
            boolean showPaths) {

        int maxFlow = 0;

        augmentationCount = 0;

        int[] parent =
                new int[vertices];

        long start =
                System.nanoTime();

        while (true) {

            for (int i = 0;
                 i < vertices;
                 i++) {

                parent[i] = -1;
            }

            boolean found =
                    bfs(
                            source,
                            sink,
                            parent);

            if (!found) {
                break;
            }

            int pathFlow =
                    Integer.MAX_VALUE;

            int current = sink;

            while (current != source) {

                int previous =
                        parent[current];

                if (residual[previous][current]
                        < pathFlow) {

                    pathFlow =
                            residual[previous][current];
                }

                current = previous;
            }

            augmentationCount++;

            if (showPaths) {

                System.out.print(
                        "\nBFS Augmenting Path "
                        + augmentationCount
                        + ": ");

                printPath(
                        source,
                        sink,
                        parent);

                System.out.println(
                        "Bottleneck Capacity : "
                        + pathFlow);
            }

            current = sink;

            while (current != source) {

                int previous =
                        parent[current];

                residual[previous][current]
                        -= pathFlow;

                residual[current][previous]
                        += pathFlow;

                current = previous;
            }

            maxFlow += pathFlow;

            if (showPaths) {

                System.out.println(
                        "Current Maximum Flow : "
                        + maxFlow);
            }
        }

        long end =
                System.nanoTime();

        return new FlowResult(
                "Edmonds-Karp",
                maxFlow,
                augmentationCount,
                end - start);
    }

    private void printPath(
            int source,
            int sink,
            int[] parent) {

        int[] path =
                new int[vertices];

        int count = 0;

        int current = sink;

        while (current != -1) {

            System.out.print(

            if (i > 0) {
                System.out.print(" -> ");
            }
        }

        System.out.println();
    }

Dinic.java

package citationflow;

public class Dinic {

    private CitationGraph graph;

    private int[][] capacity;
    private int[][] residual;

    private int vertices;

    private int[] level;
    private int[] nextEdge;

    public Dinic(CitationGraph graph) {

        this.graph = graph;

        vertices =
                graph.getPaperCount();

        capacity =
                new int[vertices][vertices];

        residual =
                new int[vertices][vertices];

        level =
                new int[vertices];

        nextEdge =
                new int[vertices];

        buildNetwork();
    }

    private void buildNetwork() {

        for (int i = 0;
             i < graph.getEdgeCount();
             i++) {

            int source =
                    graph.getInternalIndex(
                            graph.getSourceId(i));

            int target =
                    graph.getInternalIndex(
                            graph.getTargetId(i));

            capacity[source][target] += 1;
            residual[source][target] += 1;
        }
    }

    private boolean buildLevelGraph(
            int source,
            int sink) {

        for (int i = 0;
             i < vertices;
             i++) {

            level[i] = -1;
        }

        int[] queue =
                new int[vertices];

        int front = 0;
        int rear = 0;

        queue[rear++] = source;

        level[source] = 0;

        while (front < rear) {

            int current =
                    queue[front++];

            for (int next = 0;
                 next < vertices;
                 next++) {

                if (level[next] == -1
                        && residual[current][next] > 0) {

                    level[next] =
                            level[current] + 1;

                    queue[rear++] = next;
                }
            }
        }

        return level[sink] != -1;
    }

    private int sendFlow(
            int current,
            int sink,
            int flow) {

        if (current == sink) {
            return flow;
        }

        for (int next = nextEdge[current];
             next < vertices;
             next++) {

            nextEdge[current] = next;

            if (level[next]
                    == level[current] + 1
                    && residual[current][next] > 0) {

                int available =
                        Math.min(
                                flow,
                                residual[current][next]);

                int pushed =
                        sendFlow(
                                next,
                                sink,
                                available);

                if (pushed > 0) {

                    residual[current][next]
                            -= pushed;

                    residual[next][current]
                            += pushed;

                    return pushed;
                }
            }

            nextEdge[current]++;
        }

        return 0;
    }

    public FlowResult calculate(
            int source,
            int sink,
            boolean showLevels) {

        int maxFlow = 0;

        long start =
                System.nanoTime();

        int phase = 0;

        while (buildLevelGraph(
                source,
                sink)) {

            phase++;

            if (showLevels) {

                System.out.println();
                System.out.println(
                        "Level Graph " + phase);

                for (int i = 0;
                     i < vertices;
                     i++) {

                    if (level[i] != -1) {

                        System.out.println(
                                graph.getPaperId(i)
                                + " : Level "
                                + level[i]);
                    }
                }
            }

            for (int i = 0;
                 i < vertices;
                 i++) {

                nextEdge[i] = 0;
            }

            while (true) {

                int pushed =
                        sendFlow(
                                source,
                                sink,
                                Integer.MAX_VALUE);

                if (pushed == 0) {
                    break;
                }

                maxFlow += pushed;

                if (showLevels) {

                    System.out.println(
                            "Blocking flow added : "
                            + pushed);

                    System.out.println(
                            "Current Maximum Flow : "
                            + maxFlow);
                }
            }
        }

        long end =
                System.nanoTime();

        return new FlowResult(
                "Dinic",
                maxFlow,
                phase,
                end - start);
    }

    public int[][] getResidual() {
        return residual;
    }

    public int[][] getCapacity() {
        return capacity;
    }
}

    private static void minimumCutAnalysis() {

        if (!checkSourceTarget()) {
            return;
        }

        int source =
                graph.getInternalIndex(
                        sourcePaper);

        int target =
                graph.getInternalIndex(
                        targetPaper);

        FordFulkerson ff =
                new FordFulkerson(graph);

        FlowResult result =
                ff.calculate(
                        source,
                        target,
                        false);

        int[][] residual =
                ff.getResidual();

        boolean[] reachable =
                findReachableVertices(
                        source,
                        residual);

        System.out.println();
        System.out.println(
                "====================================================");

        System.out.println(
                "              MINIMUM CUT ANALYSIS");

        System.out.println(
                "====================================================");

        System.out.println(
                "\nReachable Papers:");

        for (int i = 0;
             i < graph.getPaperCount();
             i++) {

            if (reachable[i]) {

                System.out.print(
                        graph.getPaperId(i)
                        + " ");
            }
        }

        System.out.println();

        System.out.println(
                "\nCut Edges:");

        int cutCapacity = 0;

        for (int i = 0;
             i < graph.getEdgeCount();
             i++) {

            int u =
                    graph.getInternalIndex(
                            graph.getSourceId(i));

            int v =
                    graph.getInternalIndex(
                            graph.getTargetId(i));

            if (reachable[u]
                    && !reachable[v]) {

                System.out.println(
                        graph.getSourceId(i)
                        + " -> "
                        + graph.getTargetId(i)
                        + " | Capacity = 1");

                cutCapacity++;
            }
        }

        System.out.println();

        System.out.println(
                "Minimum Cut Capacity = "
                + cutCapacity);

        System.out.println(
                "Maximum Flow          = "
                + result.maxFlow);

        if (cutCapacity == result.maxFlow) {

            System.out.println();
            System.out.println(
                    "Maximum Flow = Minimum Cut");
        }

        System.out.println(
                "====================================================");
    }

    private static boolean[] findReachableVertices(
            int source,
            int[][] residual) {

        boolean[] visited =
                new boolean[
                        graph.getPaperCount()];

        int[] queue =
                new int[
                        graph.getPaperCount()];

        int front = 0;
        int rear = 0;

        queue[rear++] = source;

        visited[source] = true;

        while (front < rear) {

            int current =
                    queue[front++];

            for (int next = 0;
                 next < graph.getPaperCount();
                 next++) {

                if (!visited[next]
                        && residual[current][next] > 0) {

                    visited[next] = true;

                    queue[rear++] = next;
                }
            }
        }

        return visited;
    }
